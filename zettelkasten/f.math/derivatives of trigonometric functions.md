$$
\begin{align}
&\text{The derivative of the sine function is the cosine function:} \\[1em]
&\qquad \frac{d}{dx}(\sin x) = \cos x.
\end{align}
$$
$$
\begin{align}
&\text{And the derivative of the cosine function is the negtive of the sine function:} \\[1em]
&\qquad \frac{d}{dx}(\cos x) = \ -sin x.
\end{align}
$$

$$
\begin{align}
&\text{The derivatives of the other trigonometric functions:} \\[1.5em]
&\frac{d}{dx}(\tan x) = \sec^2 x \qquad\qquad\qquad \frac{d}{dx}(\cot x) = -\csc^2 x \\[1.5em]
&\frac{d}{dx}(\sec x) = \sec x \tan x \qquad\qquad \frac{d}{dx}(\csc x) = -\csc x \cot x
\end{align}
$$
